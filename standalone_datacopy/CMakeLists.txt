cmake_minimum_required(VERSION 3.16)
project(datacopy_data_collection LANGUAGES CXX)
set(NPU_ARCH "dav-3510" CACHE STRING "Target Ascend architecture")
set(RUN_MODE "npu" CACHE STRING "Execution target: npu or sim")
set_property(CACHE RUN_MODE PROPERTY STRINGS npu sim)
if(NOT RUN_MODE STREQUAL "npu" AND NOT RUN_MODE STREQUAL "sim")
  message(FATAL_ERROR "RUN_MODE must be npu or sim")
endif()
set(CMAKE_ASC_STANDARD 17)
set(CMAKE_ASC_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
find_package(ASC REQUIRED)
enable_language(ASC)
add_executable(demo_datacopy datacopy_data_collection.asc)
target_compile_options(demo_datacopy PRIVATE $<$<COMPILE_LANGUAGE:ASC>:--npu-arch=${NPU_ARCH}>)
if(RUN_MODE STREQUAL "sim")
  if(NOT SOC_VERSION OR NOT DEFINED ENV{ASCEND_HOME_PATH})
    message(FATAL_ERROR "sim requires SOC_VERSION and ASCEND_HOME_PATH")
  endif()
  target_link_directories(demo_datacopy PRIVATE
    "$ENV{ASCEND_HOME_PATH}/${CMAKE_SYSTEM_PROCESSOR}-linux/simulator/${SOC_VERSION}/lib")
  target_link_libraries(demo_datacopy PRIVATE runtime_camodel npu_drv)
endif()
